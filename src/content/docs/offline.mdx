<script is:inline>{`
  const CACHE_NAME = 'aipali-offline-cache';
  const MANIFEST_URL = '/assets/offline-manifest-custom.json';
  let isSyncing = false;

  // 格式化错误输出：规范全平台输出结构，消灭多重冒号
  function formatError(e) {
    if (!e) return '未知错误';
    if (typeof e === 'string') return e;
    const name = e.name && e.name !== 'Error' ? '[' + e.name + '] ' : '';
    return name + (e.message || String(e));
  }

  // 终极纯净网络请求器：绕过被壳篡改的 fetch，并坚决不带任何自定义 Header
  function safeFetch(url) {
    return new Promise(function(resolve, reject) {
      try {
        var xhr = new XMLHttpRequest();
        // 核心一：用 query 传参绕过缓存，绝不使用 setRequestHeader，避免触发壳的跨域 OPTIONS 拦截 BUG
        var reqUrl = url + (url.indexOf('?') > -1 ? '&' : '?') + '_t=' + Date.now();
        
        xhr.open('GET', reqUrl, true);
        // 核心二：使用 arraybuffer，防止在某些低版本 GeckoView 里 Blob 过 JNI 桥时崩溃
        xhr.responseType = 'arraybuffer'; 

        xhr.onload = function() {
          if (xhr.status === 0 || xhr.status >= 400) {
            reject(new Error('请求失败 (HTTP ' + xhr.status + ')：' + url));
            return;
          }
          
          var contentType = 'application/octet-stream';
          try { 
            // 只温和地尝试读取 Content-Type，不调用 getAllResponseHeaders
            var ct = xhr.getResponseHeader('Content-Type');
            if (ct) contentType = ct;
          } catch(e) {}

          try {
            // 重新组装为标准的 Response 对象供 Cache API 直接消费
            var response = new Response(xhr.response, {
              status: 200, // 强制给 Cache 200
              statusText: 'OK',
              headers: { 'Content-Type': contentType }
            });
            resolve(response);
          } catch(resErr) {
            reject(new Error('组装 Response 失败: ' + resErr.message));
          }
        };

        xhr.onerror = function() {
          reject(new Error('网络连接断开 (XHR Error) - 拦截器拒绝或断网'));
        };

        xhr.ontimeout = function() {
          reject(new Error('请求超时'));
        };

        xhr.send(null);
      } catch (err) {
        reject(err);
      }
    });
  }

  const UI = {
    log: function(msg) {
      const el = document.getElementById('log-window');
      const time = new Date().toLocaleTimeString('en-GB');
      const lines = el.innerText.split(/[\\r\\n]+/);
      if (lines.length > 80) lines.shift();
      const newline = String.fromCharCode(10);
      el.innerText = lines.join(newline) + newline + '[' + time + '] ' + msg;
      el.scrollTop = el.scrollHeight;
    },
    updateCount: function(count) {
      document.getElementById('cache-count').innerText = count;
    },
    updateStorage: function(usage) {
      document.getElementById('storage-usage').innerText = usage === '--' ? '-- MB' : usage + ' MB';
    },
    showProgress: function(show) {
      const el = document.getElementById('progress-container');
      if (show) {
        el.classList.remove('hidden');
        void el.offsetWidth;
        el.classList.add('visible');
      } else {
        el.classList.remove('visible');
        setTimeout(function() { el.classList.add('hidden'); }, 300);
      }
    },
    showTerminal: function(show) {
      const el = document.getElementById('terminal-window');
      if (!el) return;
      if (show) {
        el.classList.remove('hidden');
        void el.offsetWidth;
        el.classList.add('visible');
      } else {
        el.classList.remove('visible');
        setTimeout(function() { el.classList.add('hidden'); }, 300);
      }
    },
    setProgress: function(current, total, text) {
      const percent = total > 0 ? Math.round((current / total) * 100) : 0;
      document.getElementById('progress-bar-fill').style.width = percent + '%';
      document.getElementById('progress-percent').innerText = percent + '%';
      if (text) document.getElementById('progress-text').innerText = text;
    },
    disableButtons: function(disabled) {
      document.getElementById('btn-update').disabled = disabled;
      document.getElementById('btn-rebuild').disabled = disabled;
      document.getElementById('btn-clear').disabled = disabled;
    }
  };

  // 通用稳健清理：兼容全平台环境及文件占用场景
  async function safeClearAllCaches() {
    if (!('caches' in window)) return;
    try {
      const cacheNames = await caches.keys();
      for (const name of cacheNames) {
        UI.log('正在清理缓存分区 (' + name + ')...');
        let deleted = false;
        try {
          deleted = await caches.delete(name);
        } catch (delErr) {}

        // 降级容错：若直接移除失败，则清空分区内部条目，确保数据完全释放
        if (!deleted) {
          try {
            const c = await caches.open(name);
            const keys = await c.keys();
            for (const req of keys) {
              try { await c.delete(req); } catch(kErr) {}
            }
            UI.log('已完成分区数据清空 (' + name + ')');
          } catch (fallbackErr) {
            UI.log('清理缓存分区 (' + name + ') 遇到问题 - ' + formatError(fallbackErr));
          }
        }
      }
    } catch (e) {
      UI.log('读取缓存列表失败 - ' + formatError(e));
    }
  }

  async function refreshStatus() {
    try {
      if ('caches' in window) {
        const cacheNames = await caches.keys();
        const pathSet = new Set(); 
        
        for (let name of cacheNames) {
          try {
            const cache = await caches.open(name);
            const keys = await cache.keys();
            
            for (let req of keys) {
              try {
                const url = new URL(req.url);
                const path = url.pathname;
                
                const isPageNode = path.endsWith('/') || path.endsWith('.html');
                const isManifestNode = path === MANIFEST_URL;
                const isAssetNode = path.startsWith('/assets/') || path.startsWith('/_astro/'); 
                const isInternalJunk = path.includes('/@vite/') || path.includes('/@fs/');

                if ((isPageNode || isManifestNode || isAssetNode) && !isInternalJunk) {
                  pathSet.add(path);
                }
              } catch(e) {}
            }
          } catch(cacheOpenErr) {}
        }
        UI.updateCount(pathSet.size);
      } else {
        UI.updateCount('N/A');
        UI.log('系统提示：当前环境不支持 Cache API，离线功能不可用。');
      }

      if (navigator.storage && navigator.storage.estimate) {
        const estimate = await navigator.storage.estimate();
        const usedMB = (estimate.usage / (1024 * 1024)).toFixed(2);
        UI.updateStorage(usedMB);
      } else {
        UI.updateStorage('--');
      }
    } catch (e) {
      UI.log('读取存储状态失败 - ' + formatError(e));
    }
  }

  function refreshStatusWithRetry() {
    refreshStatus();
    setTimeout(refreshStatus, 1000);
    setTimeout(refreshStatus, 3000);
  }

  window.confirmAndClear = async function() {
    if (isSyncing) return;
    const msg = "清除本地数据：\\n\\n此操作将彻底清除所有本地缓存数据。清除后，应用将不再提供离线阅读功能，并恢复到初始化状态。后续您可以通过全量重新构建，来重新开启离线阅读功能。\\n\\n确认执行本地数据清除？";
    if (!confirm(msg)) return;
    
    UI.showTerminal(true);
    UI.log('正在执行本地数据清理...');
    UI.disableButtons(true);
    try {
      if ('serviceWorker' in navigator) {
        try {
          const registrations = await navigator.serviceWorker.getRegistrations();
          for (const reg of registrations) {
            try { await reg.unregister(); } catch(swErr) {}
          }
        } catch(swTotalErr) {}
      }

      await safeClearAllCaches();
      
      if (window.indexedDB && indexedDB.databases) {
        try {
          const dbs = await indexedDB.databases();
          for (const db of dbs) {
            if (db && db.name) {
              try {
                const req = indexedDB.deleteDatabase(db.name);
                await new Promise(function(resolve) {
                  req.onsuccess = function() { resolve(); };
                  req.onerror = function() { resolve(); };
                  req.onblocked = function() { resolve(); };
                  setTimeout(resolve, 1000);
                });
              } catch(dbErr) {}
            }
          }
        } catch(err) {}
      }

      UI.log('本地离线经文数据与相关缓存已全部清除。');
      UI.log('提示：建议[彻底划掉/退出]当前应用并重新进入，以刷新准确的存储占用数值。');
      
    } catch (e) {
      UI.log('清除数据失败 - ' + formatError(e));
    } finally {
      UI.disableButtons(false);
      UI.updateCount('--');
      UI.updateStorage('--');
      refreshStatusWithRetry();
    }
  };

  window.confirmAndStartSync = async function(isRebuild) {
    if (isSyncing) return;
    if (!navigator.onLine) {
      alert("网络未连接\\n\\n设备当前处于离线状态，无法连接至云端服务器。请在网络连接恢复后再试。");
      return;
    }

    const msg = isRebuild
      ? "全量重新构建\\n\\n系统将清除现有数据，并从云端完整拉取所有经文节点。\\n\\n确认执行全量重建？"
      : "增量同步数据\\n\\n系统将比对本地与云端目录，自动下载新增或变动节点。\\n\\n确认执行增量同步？";
    
    if (!confirm(msg)) return;
    UI.showTerminal(true);
    await performSync(isRebuild);
  };

  async function performSync(isRebuild) {
    isSyncing = true;
    UI.disableButtons(true);
    UI.showProgress(true);
    UI.showTerminal(true);
    
    UI.log('');
    UI.log(isRebuild ? '=== 开始全量重新构建 ===' : '=== 开始增量同步数据 ===');

    try {
      UI.log('正在获取云端经文目录 (sitemap.xml)...');
      // 使用降级后的 safeFetch 强制请求
      const sitemapRes = await safeFetch('/sitemap-0.xml');
      if (!sitemapRes.ok) throw new Error('云端服务器通信异常 (HTTP ' + sitemapRes.status + ')');
      
      const sitemapText = await sitemapRes.text();
      const parser = new DOMParser();
      const xml = parser.parseFromString(sitemapText, "text/xml");
      
      const urls = Array.from(xml.querySelectorAll("loc"))
        .map(function(node) {
          try { return new URL(node.textContent).pathname; } 
          catch(e) { return node.textContent; }
        })
        .filter(function(url) {
          return !url.includes('/tags/');
        });

      // ==========================================================
      // 1. 读取额外静态数据清单
      // ==========================================================
      try {
        UI.log('正在读取离线资源清单 (manifest)...');
        const manifestRes = await safeFetch(MANIFEST_URL);
        if (manifestRes.ok) {
          // 由于 Response 内部基于 ArrayBuffer，需要反序列化为 JSON
          const manifestText = await manifestRes.text();
          const manifest = JSON.parse(manifestText);
          if (manifest && Array.isArray(manifest.nodes)) {
            urls.push(...manifest.nodes);
            urls.push(MANIFEST_URL);
            UI.log('已加载清单扩展资源：共 ' + manifest.nodes.length + ' 个节点');
          }
        }
      } catch (err) {
        UI.log('未能加载额外资源清单，跳过。');
      }

      // ==========================================================
      // 2. 自动从 DOM 抓取页面依赖资源
      // ==========================================================
      try {
        UI.log('正在解析页面依赖资源 (样式、脚本与图标)...');
        const assetElements = document.querySelectorAll(
          'link[rel="stylesheet"], link[rel="preload"][as="style"], link[rel="preload"][as="font"], link[rel="apple-touch-icon"], link[rel="icon"], script[src], img[src]'
        );
        
        const autoAssets = [];
        for (const el of assetElements) {
          const rawUrl = el.src || el.href;
          if (!rawUrl || rawUrl.startsWith('data:')) continue;
          
          try {
            const parsed = new URL(rawUrl, window.location.origin);
            if (parsed.origin === window.location.origin) {
              if (parsed.pathname.includes('/_astro/') || parsed.pathname.includes('/assets/')) {
                autoAssets.push(parsed.pathname);
              }
            }
          } catch (e) {}
        }

        const uniqueAssets = Array.from(new Set(autoAssets));
        let addedCount = 0;
        for (const assetUrl of uniqueAssets) {
          if (!urls.includes(assetUrl)) {
            urls.push(assetUrl);
            addedCount++;
          }
        }
        UI.log('已载入页面核心资源：共 ' + addedCount + ' 个');
      } catch (err) {}

      if (urls.length === 0) throw new Error('未获取到有效的同步节点');
      UI.log('节点校验完成：待处理节点共计 ' + urls.length + ' 个');

      let existingKeys = [];
      if (isRebuild) {
        UI.log('正在重置本地旧版缓存...');
        await safeClearAllCaches();
      } else {
        try {
          if ('caches' in window) {
            const checkCache = await caches.open(CACHE_NAME);
            const reqs = await checkCache.keys();
            existingKeys = reqs.map(function(req) { return new URL(req.url).pathname; });
            UI.log('正在比对，跳过本地已存在的节点...');
          }
        } catch (err) {
          existingKeys = [];
        }
      }

      // 打开统一缓存空间
      const targetCache = await caches.open(CACHE_NAME);

      let downloaded = 0;
      let skipped = 0;
      let errors = 0;
      let currentIndex = 0;
      const MAX_CONCURRENT = 5;

      const downloadWorker = async function() {
        while (currentIndex < urls.length) {
          const index = currentIndex++;
          const url = urls[index];
          
          let displayUrl = url;
          if (url.length > 36) {
            displayUrl = url.substring(0, 16) + '...' + url.substring(url.length - 16);
          }

          if (!isRebuild && existingKeys.includes(url)) {
            skipped++;
          } else {
            try {
              UI.setProgress(currentIndex, urls.length, '正在下载：' + displayUrl);
              const res = await safeFetch(url);
              
              if (res.ok) {
                // 注意：Cache 存入时的键名依然是原始的、没有时间戳的 url
                await targetCache.put(url, res);
                downloaded++;
              } else {
                UI.log('下载响应异常 (HTTP ' + res.status + ')：' + url);
                errors++;
              }
            } catch (err) {
              UI.log('写入缓存失败 (' + url + ') - ' + formatError(err));
              errors++;
            }
          }
          
          if (currentIndex % 10 === 0 || currentIndex >= urls.length - MAX_CONCURRENT) {
            UI.setProgress(currentIndex, urls.length); 
          }
        }
      };

      UI.log('启动多线程并行传输 (并发数：5)...');
      const workers = Array(Math.min(MAX_CONCURRENT, urls.length)).fill(null).map(function() { return downloadWorker(); });
      await Promise.all(workers);

      UI.setProgress(urls.length, urls.length, '同步完成');
      UI.log('同步任务完成：新增/更新 ' + downloaded + ' 个，已跳过 ' + skipped + ' 个，失败 ' + errors + ' 个');

    } catch (e) {
      UI.log('同步中断 - ' + formatError(e));
    } finally {
      isSyncing = false;
      UI.disableButtons(false);
      setTimeout(function() { UI.showProgress(false); }, 3000);
      refreshStatusWithRetry();
    }
  }

  document.addEventListener('DOMContentLoaded', function() {
    refreshStatus();
  });
`}</script>
