<script is:inline>{`
(function(){
  var c = document.getElementById('aipali-particles');
  if (!c) return;
  var ctx = c.getContext('2d');
  var W, H;
  function resize(){ W = c.width = window.innerWidth; H = c.height = window.innerHeight; }
  resize(); window.addEventListener('resize', resize);

  var count = Math.min(70, Math.floor(W * H / 15000));
  var dots = Array.from({length: count}, function(){
    return {
      x: Math.random() * W, y: Math.random() * H,
      vx: (Math.random() - 0.5) * 0.4, vy: (Math.random() - 0.5) * 0.4,
      r: Math.random() * 2 + 0.5
    };
  });

  function isDark(){ return document.documentElement.getAttribute('data-theme') === 'dark' || document.documentElement.classList.contains('dark') || window.matchMedia('(prefers-color-scheme: dark)').matches; }

  var observer = new MutationObserver(function(){ draw(); });
  observer.observe(document.documentElement, {attributes: true, attributeFilter: ['data-theme', 'class']});

  function draw(){
    ctx.clearRect(0, 0, W, H);
    var alpha = isDark() ? 0.18 : 0.10;
    var color = isDark() ? '200,170,255' : '100,80,180';
    for (var i = 0; i < dots.length; i++){
      var d = dots[i];
      d.x += d.vx; d.y += d.vy;
      if (d.x < 0 || d.x > W) d.vx *= -1;
      if (d.y < 0 || d.y > H) d.vy *= -1;
      ctx.beginPath(); ctx.arc(d.x, d.y, d.r, 0, Math.PI * 2);
      ctx.fillStyle = 'rgba(' + color + ',' + (alpha * 1.5) + ')';
      ctx.fill();
    }
    for (var i = 0; i < dots.length; i++){
      for (var j = i + 1; j < dots.length; j++){
        var dx = dots[i].x - dots[j].x, dy = dots[i].y - dots[j].y;
        var dist = Math.sqrt(dx * dx + dy * dy);
        if (dist < 150){
          ctx.beginPath(); ctx.moveTo(dots[i].x, dots[i].y); ctx.lineTo(dots[j].x, dots[j].y);
          ctx.strokeStyle = 'rgba(' + color + ',' + (alpha * (1 - dist / 150)) + ')';
          ctx.lineWidth = 0.5; ctx.stroke();
        }
      }
    }
    requestAnimationFrame(draw);
  }
  draw();
})();
`}</script>
